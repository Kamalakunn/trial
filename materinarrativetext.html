/** * 🏆 Leaderboard Backend for Kingdom of Narrata * * CARA SETUP: * 1. Buka Google Sheets baru * 2. Extensions → Apps Script * 3. Paste kode ini * 4. Deploy → New deployment → Web app * - Execute as: Me * - Who has access: Anyone * 5. Copy URL yang muncul * 6. Paste URL tersebut ke variabel Leaderboard.ENDPOINT di index.html */ const SHEET_NAME = 'Leaderboard'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = getOrCreateSheet(); sheet.appendRow([ new Date(), data.name || 'Hero', data.class || '-', data.xp || 0, data.crystals || 0, data.hp || 0 ]); return ContentService .createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const action = e.parameter.action; if (action === 'getTop') { const sheet = getOrCreateSheet(); const data = sheet.getDataRange().getValues(); // Skip header row const scores = data.slice(1).map(row => ({ date: row[0], name: row[1], class: row[2], xp: row[3], crystals: row[4], hp: row[5] })); // Sort by XP descending, take top 10 scores.sort((a, b) => b.xp - a.xp); const top10 = scores.slice(0, 10); return ContentService .createTextOutput(JSON.stringify({ success: true, scores: top10 })) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(JSON.stringify({ success: true, message: 'API ready' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function getOrCreateSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['Timestamp', 'Name', 'Class', 'XP', 'Crystals', 'HP']); sheet.getRange(1, 1, 1, 6).setFontWeight('bold').setBackground('#6a4cff').setFontColor('#ffffff'); sheet.setFrozenRows(1); } return sheet; } // Test function - run this manually to verify setup function testSetup() { const sheet = getOrCreateSheet(); Logger.log('Sheet ready: ' + sheet.getName()); Logger.log('Deploy this script as Web App to get your endpoint URL!'); }
